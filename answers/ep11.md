# 11회 정답 — CSV 가져오기와 숨은 결함

> 시작 브랜치는 `start/ep11`입니다. 빨강 3개로 시작합니다.

## 이 실습의 함정

`src/csv.js`에는 문제가 **두 개** 있습니다.

1. **가져오기가 아예 없다** (`fromCsv`가 `TODO`)
2. **내보내기가 쉼표를 처리하지 않는다** — 이건 가져오기를 만들기 전까지 **드러나지 않습니다.**

메모에 쉼표가 들어 있으면(`"점심, 커피"`) 내보낸 CSV의 칸이 밀립니다.
가져오기를 아무리 잘 만들어도, 내보내기를 안 고치면 왕복이 성립하지 않습니다.

## 정답 코드

```js
// 값 하나를 CSV 칸으로 바꾼다. 쉼표·따옴표·줄바꿈이 있으면 따옴표로 감싼다.
function toCsvCell(value) {
  var text = String(value);
  if (/[",\n]/.test(text)) {
    return '"' + text.replace(/"/g, '""') + '"';
  }
  return text;
}

// CSV 한 줄을 칸 배열로 쪼갠다. 따옴표 안의 쉼표는 칸을 나누지 않는다.
function splitCsvLine(line) {
  var cells = [];
  var current = '';
  var inQuotes = false;
  for (var i = 0; i < line.length; i++) {
    var ch = line[i];
    if (inQuotes) {
      if (ch === '"' && line[i + 1] === '"') { current += '"'; i++; }
      else if (ch === '"') { inQuotes = false; }
      else { current += ch; }
    } else if (ch === '"') {
      inQuotes = true;
    } else if (ch === ',') {
      cells.push(current); current = '';
    } else {
      current += ch;
    }
  }
  cells.push(current);
  return cells;
}

// CSV 문자열을 항목 배열로 바꾼다.
function fromCsv(text) {
  var lines = String(text).replace(/^﻿/, '').split(/\r?\n/);
  var items = [];
  for (var i = 0; i < lines.length; i++) {
    if (lines[i].trim() === '') { continue; }
    var cells = splitCsvLine(lines[i]);
    if (i === 0 && cells[0] === CSV_HEADER[0]) { continue; }   // 머리글 줄은 건너뛴다
    items.push({
      id: 'e' + String(items.length + 1).padStart(3, '0'),
      date: cells[0],
      category: cells[1],
      amount: Number(cells[2]),
      memo: cells[3] === undefined ? '' : cells[3]
    });
  }
  return items;
}
```
